CREATE DATABASE IF NOT EXISTS bookstore
CHARACTER SET utf8
COLLATE utf8_general_ci;

USE bookstore;

SET NAMES utf8;

-- Роли
CREATE TABLE user_roles (
    role_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(23) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ФИО
CREATE TABLE user_fio (
    fio_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Пользователи
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    role INT NOT NULL,
    fio INT NOT NULL,
    login VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(10) NOT NULL,
    FOREIGN KEY (role) REFERENCES user_roles(role_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (fio) REFERENCES user_fio(fio_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Категории
CREATE TABLE categories (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Производители (издательства)
CREATE TABLE manufacturers (
    manufacturer_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Поставщики (авторы)
CREATE TABLE suppliers (
    supplier_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Единицы измерения
CREATE TABLE unit (
    unit_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(5) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Товары (книги)
CREATE TABLE tovar (
    article VARCHAR(6) NOT NULL PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    unit_id INT NOT NULL DEFAULT 1,
    price INT NOT NULL,
    supplier_id INT NOT NULL,
    manufacturer_id INT NOT NULL,
    category_id INT NOT NULL,
    discount INT DEFAULT 0,
    stock_quantity INT DEFAULT 0,
    description VARCHAR(500),
    image_path VARCHAR(200),
    FOREIGN KEY (unit_id) REFERENCES unit(unit_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (supplier_id) REFERENCES suppliers(supplier_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (manufacturer_id) REFERENCES manufacturers(manufacturer_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (category_id) REFERENCES categories(category_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Статусы заказов
CREATE TABLE order_status (
    status_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(10) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ПВЗ
CREATE TABLE pvz (
    pvz_id INT AUTO_INCREMENT PRIMARY KEY,
    address VARCHAR(100) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Заказы
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    order_date DATE,
    delivery_date DATE,
    pvz_id INT NOT NULL,
    client_fio_id INT,
    pickup_code INT,
    status_id INT NOT NULL,
    FOREIGN KEY (pvz_id) REFERENCES pvz(pvz_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (client_fio_id) REFERENCES user_fio(fio_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (status_id) REFERENCES order_status(status_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Состав заказов (3НФ)
CREATE TABLE order_items (
    order_item_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    article VARCHAR(6) NOT NULL,
    quantity INT NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (article) REFERENCES tovar(article) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
